Applications of Stack

1. Function calls and recursion
    The call stack stores function parameters, local variables, and return addresses.
    It allows recursive functions to return to the correct calling function.

2. Expression conversion and evaluation
    Stacks are used to convert infix expressions to postfix or prefix notation and
    to evaluate postfix and prefix expressions.

3. Parentheses and syntax checking
    An opening bracket is pushed onto a stack. When a closing bracket is found, the
    top opening bracket is popped and checked for a matching pair.

4. Undo and redo operations
    Applications store recent actions in stacks. One stack supports undo, while a
    second stack can store actions that have been undone for redo operations.

5. Backtracking
    Stacks store previous choices in problems such as maze solving, depth-first
    search, and the N-Queens problem so that a previous state can be restored.
#include <stdio.h>
#include <string.h>

#define SIZE 100

int is_matching(char opening, char closing) {
    return (opening == '(' && closing == ')') ||
           (opening == '[' && closing == ']') ||
           (opening == '{' && closing == '}');
}

int balanced_parentheses(const char expression[]) {
    char stack[SIZE];
    int top = -1;
    int index;

    for (index = 0; expression[index] != '\0'; index++) {
        char character = expression[index];

        if (character == '(' || character == '[' || character == '{') {
            if (top == SIZE - 1) {
                return 0;
            }
            stack[++top] = character;
        } else if (character == ')' || character == ']' || character == '}') {
            if (top == -1 || !is_matching(stack[top], character)) {
                return 0;
            }
            top--;
        }
    }

    return top == -1;
}

void reverse_string(char text[]) {
    char stack[SIZE];
    int top = -1;
    int index;

    for (index = 0; text[index] != '\0'; index++) {
        stack[++top] = text[index];
    }
    for (index = 0; top >= 0; index++) {
        text[index] = stack[top--];
    }
}

int main(void) {
    char expression[SIZE];
    char text[SIZE];
    int choice;

    do {
        printf("\n1. Check balanced parentheses\n2. Reverse text\n3. Exit\nChoice: ");
        scanf("%d", &choice);
        getchar();

        switch (choice) {
            case 1:
                printf("Enter an expression: ");
                fgets(expression, SIZE, stdin);
                expression[strcspn(expression, "\n")] = '\0';
                printf(balanced_parentheses(expression)
                           ? "Parentheses are balanced.\n"
                           : "Parentheses are not balanced.\n");
                break;
            case 2:
                printf("Enter text: ");
                fgets(text, SIZE, stdin);
                text[strcspn(text, "\n")] = '\0';
                reverse_string(text);
                printf("Reversed text: %s\n", text);
                break;
            case 3:
                break;
            default:
                printf("Invalid choice.\n");
        }
    } while (choice != 3);

    return 0;
}
